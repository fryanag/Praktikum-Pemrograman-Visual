Public Class Form1

    '==================================================
    ' FORM LOAD
    '==================================================
    Private Sub Form1_Load(sender As Object, e As EventArgs) Handles MyBase.Load

        'Isi pilihan jenis
        cmbJenis.Items.Add("Karyawan")
        cmbJenis.Items.Add("Mahasiswa")

        'Sembunyikan input NIP dan NIM saat awal
        txtNIP.Visible = False
        txtNIM.Visible = False

        'Pajak awal
        lblPajak.Text = "Rp 0"

    End Sub


    '==================================================
    ' PILIH JENIS
    '==================================================
    Private Sub cmbJenis_SelectedIndexChanged(
        sender As Object,
        e As EventArgs
    ) Handles cmbJenis.SelectedIndexChanged

        'Bersihkan input
        txtNama.Clear()
        txtNIP.Clear()
        txtNIM.Clear()
        txtPenghasilan.Clear()

        'Reset pajak
        lblPajak.Text = "Rp 0"


        'Jika memilih Karyawan
        If cmbJenis.Text = "Karyawan" Then

            txtNIP.Visible = True
            txtNIM.Visible = False

            'Gambar karyawan
            picImage.Image = Image.FromFile("Assets\karyawan.png")


        'Jika memilih Mahasiswa
        ElseIf cmbJenis.Text = "Mahasiswa" Then

            txtNIP.Visible = False
            txtNIM.Visible = True

            'Gambar mahasiswa
            picImage.Image = Image.FromFile("Assets\mahasiswa.png")

        End If

    End Sub


    '==================================================
    ' BUTTON HITUNG
    '==================================================
    Private Sub btnHitung_Click(
        sender As Object,
        e As EventArgs
    ) Handles btnHitung.Click

        'Variabel
        Dim penghasilan As Decimal
        Dim pajak As Decimal


        '==================================================
        ' VALIDASI JENIS
        '==================================================
        If cmbJenis.Text = "" Then

            MessageBox.Show("Pilih jenis terlebih dahulu")
            cmbJenis.Focus()
            Return

        End If


        '==================================================
        ' VALIDASI NAMA
        '==================================================
        If txtNama.Text = "" Then

            MessageBox.Show("Nama belum diisi")
            txtNama.Focus()
            Return

        End If


        '==================================================
        ' VALIDASI NIP UNTUK KARYAWAN
        '==================================================
        If cmbJenis.Text = "Karyawan" Then

            If txtNIP.Text = "" Then

                MessageBox.Show("NIP belum diisi")
                txtNIP.Focus()
                Return

            End If

        End If


        '==================================================
        ' VALIDASI NIM UNTUK MAHASISWA
        '==================================================
        If cmbJenis.Text = "Mahasiswa" Then

            If txtNIM.Text = "" Then

                MessageBox.Show("NIM belum diisi")
                txtNIM.Focus()
                Return

            End If

        End If


        '==================================================
        ' VALIDASI PENGHASILAN
        '==================================================
        If Not Decimal.TryParse(
            txtPenghasilan.Text,
            penghasilan
        ) Then

            MessageBox.Show("Masukkan penghasilan dalam bentuk angka")
            txtPenghasilan.Focus()
            Return

        End If


        '==================================================
        ' VALIDASI NILAI PENGHASILAN
        '==================================================
        If penghasilan < 0 Then

            MessageBox.Show("Penghasilan tidak boleh kurang dari 0")
            txtPenghasilan.Focus()
            Return

        End If


        '==================================================
        ' PERHITUNGAN PAJAK
        '==================================================

        'Jika penghasilan 0 maka pajak 0
        If penghasilan = 0 Then

            pajak = 0

        Else

            '==================================================
            ' RUMUS PAJAK
            '==================================================
            ' GANTI BAGIAN INI SESUAI RUMUS DARI DOSEN
            '
            ' CONTOH:
            ' Dim tarifPajak As Decimal = 0.05D
            ' pajak = penghasilan * tarifPajak
            '
            '==================================================

            Dim tarifPajak As Decimal = 0.05D

            pajak = penghasilan * tarifPajak

        End If


        '==================================================
        ' TAMPILKAN PAJAK
        '==================================================
        lblPajak.Text = "Rp " & pajak.ToString("N0")

    End Sub


    '==================================================
    ' TEXTBOX PENGHASILAN - HANYA ANGKA
    '==================================================
    Private Sub txtPenghasilan_KeyPress(
        sender As Object,
        e As KeyPressEventArgs
    ) Handles txtPenghasilan.KeyPress

        'Hanya menerima angka dan tombol kontrol
        If Not Char.IsControl(e.KeyChar) AndAlso
           Not Char.IsDigit(e.KeyChar) Then

            e.Handled = True

        End If

    End Sub

End Class
